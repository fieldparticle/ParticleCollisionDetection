135class Tokenizer:
136 '''Tokenize an input string
137
138 Typical usage:
139
140 tokens = list(Tokenizer("<memory>").tokenize("""a = 7; b = ();"""))
141
142 The filename argument to the constructor is used only in error messages, no
143 data is loaded from the file. The input data is received as argument to the
144 tokenize function, which yields tokens or throws a ConfigParseError on
145 invalid input.
146
147 Include directives are not supported, they must be handled at a higher
148 level (cf. the TokenStream class).
149 '''
150
151 token_map = compile_regexes([
152 (FltToken, 'float', r'([-+]?(\d+)?\.\d*([eE][-+]?\d+)?)|'
153 r'([-+]?(\d+)(\.\d*)?[eE][-+]?\d+)'),
154 (IntToken, 'hex64', r'0[Xx][0-9A-Fa-f]+(L(L)?)'),
155 (IntToken, 'hex', r'0[Xx][0-9A-Fa-f]+'),
156 (IntToken, 'integer64', r'[-+]?[0-9]+L(L)?'),
157 (IntToken, 'integer', r'[-+]?[0-9]+'),
158 (BoolToken, 'boolean', r'(?i)(true|false)\b'),
159 (StrToken, 'string', r'"([^"\\]|\\.)*"'),
160 (Token, 'name', r'[A-Za-z\*][-A-Za-z0-9_\*]*'),
161 (Token, '}', r'\}'),
162 (Token, '{', r'\{'),
163 (Token, ')', r'\‍)'),
164 (Token, '(', r'\‍('),
165 (Token, ']', r'\]'),
166 (Token, '[', r'\['),
167 (Token, ',', r','),
168 (Token, ';', r';'),
169 (Token, '=', r'='),
170 (Token, ':', r':'),
171 ])
172
173 def __init__(self, filename):
174 self.filename = filename
175 self.row = 1
176 self.column = 1
177
178 def tokenize(self, string):
179 '''Yield tokens from the input string or throw ConfigParseError'''
180 pos = 0
181 while pos < len(string):
182 m = SKIP_RE.match(string, pos=pos)
183 if m:
184 skip_lines = m.group(0).split('\n')
185 if len(skip_lines) > 1:
186 self.row += len(skip_lines) - 1
187 self.column = 1 + len(skip_lines[-1])
188 else:
189 self.column += len(skip_lines[0])
190
191 pos = m.end()
192 continue
193
194 for cls, type, regex in self.token_map:
195 m = regex.match(string, pos=pos)
196 if m:
197 yield cls(type, m.group(0),
198 self.filename, self.row, self.column)
199 self.column += len(m.group(0))
200 pos = m.end()
201 break
202 else:
203 raise ConfigParseError(
204 "Couldn't load config in %r row %d, column %d: %r" %
205 (self.filename, self.row, self.column,
206 string[pos:pos+20]))

209class TokenStream:
210 '''Offer a parsing-oriented view on tokens
211
212 Provide several methods that are useful to parsers, like ``accept()``,
213 ``expect()``, ...
214
215 The ``from_file()`` method is the preferred way to read input files, as
216 it handles include directives, which the ``Tokenizer`` class does not do.
217 '''
218
219 def __init__(self, tokens):
220 self.position = 0
221 self.tokens = list(tokens)
222
223 @classmethod
224 def from_file(cls, f, filename=None, includedir='', seenfiles=None):
225 '''Create a token stream by reading an input file
226
227 Read tokens from `f`. If an include directive ('@include "file.cfg"')
228 is found, read its contents as well.
229
230 The `filename` argument is used for error messages and to detect
231 circular imports. ``includedir`` sets the lookup directory for included
232 files. ``seenfiles`` is used internally to detect circular includes,
233 and should normally not be supplied by users of is function.
234 '''
235
236 if filename is None:
237 filename = getattr(f, 'name', '<unknown>')
238 if seenfiles is None:
239 seenfiles = set()
240
241 if filename in seenfiles:
242 raise ConfigParseError("Circular include: %r" % (filename,))
243 seenfiles = seenfiles | {filename} # Copy seenfiles, don't alter it.
244
245 tokenizer = Tokenizer(filename=filename)
246 lines = []
247 tokens = []
248 for line in f:
249 m = re.match(r'@include "(.*)"$', line.strip())
250 if m:
251 tokens.extend(tokenizer.tokenize(''.join(lines)))
252 lines = [re.sub(r'\S', ' ', line)]
253
254 includefilename = decode_escapes(m.group(1))
255 includefilename = os.path.join(includedir, includefilename)
256 try:
257 includefile = open(includefilename, "r")
258 except IOError:
259 raise ConfigParseError("Could not open include file %r" %
260 (includefilename,))
261
262 with includefile:
263 includestream = cls.from_file(includefile,
264 filename=includefilename,
265 includedir=includedir,
266 seenfiles=seenfiles)
267 tokens.extend(includestream.tokens)
268
269 else:
270 lines.append(line)
271
272 tokens.extend(tokenizer.tokenize(''.join(lines)))
273 return cls(tokens)

275 def peek(self):
276 '''Return (but do not consume) the next token
277
278 At the end of input, ``None`` is returned.
279 '''
280
281 if self.position >= len(self.tokens):
282 return None
283
284 return self.tokens[self.position]

286 def accept(self, *args):
287 '''Consume and return the next token if it has the correct type
288
289 Multiple token types (as strings, e.g. 'integer64') can be given
290 as arguments. If the next token is one of them, consume and return it.
291
292 If the token type doesn't match, return None.
293 '''
294
295 token = self.peek()
296 if token is None:
297 return None
298
299 for arg in args:
300 if token.type == arg:
301 self.position += 1
302 return token
303
304 return None
305
306 def expect(self, *args):
307 '''Consume and return the next token if it has the correct type
308
309 Multiple token types (as strings, e.g. 'integer64') can be given
310 as arguments. If the next token is one of them, consume and return it.
311
312 If the token type doesn't match, raise a ConfigParseError.
313 '''
314
315 t = self.accept(*args)
316 if t is not None:
317 return t
318
319 self.error("expected: %r" % (args,))
320
321 def error(self, msg):
322 '''Raise a ConfigParseError at the current input position'''
323 if self.finished():
324 raise ConfigParseError("Unexpected end of input; %s" % (msg,))
325 else:
326 t = self.peek()
327 raise ConfigParseError("Unexpected token %s; %s" % (t, msg))

329 def finished(self):
330 '''Return ``True`` if the end of the token stream is reached.'''
331 return self.position >= len(self.tokens)

336
337 Takes a ``TokenStream`` as input, the ``parse()`` method then returns

347 def configuration(self):
348 result = self.setting_list_or_empty()
349 if not self.tokens.finished():
350 raise ConfigParseError("Expected end of input but found %s" %
351 (self.tokens.peek(),))

355 def setting_list_or_empty(self):
356 result = AttrDict()
357 while True:
358 s = self.setting()
359 if s is None:

364 def setting(self):
365 name = self.tokens.accept('name')
366 if name is None:
367 return None
368
369 self.tokens.expect(':', '=')
370
371 value = self.value()
372 if value is None:
373 self.tokens.error("expected a value")
374
375 self.tokens.accept(';', ',')

379 def value(self):
380 acceptable = [self.scalar_value, self.array, self.list, self.group]
381 return self._parse_any_of(acceptable)

383 def scalar_value(self):
384 # This list is ordered so that more common tokens are checked first.
385 acceptable = [self.string, self.boolean, self.integer, self.float,
386 self.hex, self.integer64, self.hex64]
387 return self._parse_any_of(acceptable)

389 def value_list_or_empty(self):
390 return tuple(self._comma_separated_list_or_empty(self.value))

392 def scalar_value_list_or_empty(self):
393 return self._comma_separated_list_or_empty(self.scalar_value)

395 def array(self):
396 return self._enclosed_block('[', self.scalar_value_list_or_empty, ']')

398 def list(self):

422 def string(self):
423 t_first = self.tokens.accept('string')
424 if t_first is None:
425 return None
426
427 values = [t_first.value]

437 t = self.tokens.accept(tokentype)

443 def _parse_any_of(self, nonterminals):
444 for fun in nonterminals:

452 values = []

549def dump_string(s):
550 '''Stringize ``s``, adding double quotes and escaping as necessary
551
552 Backslash escape backslashes, double quotes, ``\f``, ``\n``, ``\r``, and
553 ``\t``. Escape all remaining unprintable characters in ``\xFF``-style.
554 The returned string will be surrounded by double quotes.
555 '''
556
557 s = (s.replace('\\', '\\\\')
558 .replace('"', '\\"')
559 .replace('\f', r'\f')
560 .replace('\n', r'\n')
561 .replace('\r', r'\r')
562 .replace('\t', r'\t'))
563 s = UNPRINTABLE_CHARACTER_RE.sub(
564 lambda m: r'\x{:02x}'.format(ord(m.group(0))),
565 s)
566 return '"' + s + '"'

569def get_dump_type(value):
570 '''Get the libconfig datatype of a value
571
572 Return values: ``'d'`` (dict), ``'l'`` (list), ``'a'`` (array),
573 ``'i'`` (integer), ``'i64'`` (long integer), ``'b'`` (bool),
574 ``'f'`` (float), or ``'s'`` (string).
575
576 Produces the proper type for LibconfList, LibconfArray, LibconfInt64
577 instances.
578 '''
579
580 if isinstance(value, dict):
581 return 'd'
582 if isinstance(value, tuple):
583 return 'l'
584 if isinstance(value, list):
585 return 'a'
586
587 # Test bool before int since isinstance(True, int) == True.
588 if isinstance(value, bool):
589 return 'b'
590 if isint(value):
591 if is_long_int(value):
592 return 'i64'
593 else:
594 return 'i'
595 if isinstance(value, float):
596 return 'f'
597 if isstr(value):
598 return 's'
599
600 return None

603def get_array_value_dtype(lst):
604 '''Return array value type, raise ConfigSerializeError for invalid arrays
605
606 Libconfig arrays must only contain scalar values and all elements must be
607 of the same libconfig data type. Raises ConfigSerializeError if these
608 invariants are not met.
609
610 Returns the value type of the array. If an array contains both int and
611 long int data types, the return datatype will be ``'i64'``.
612 '''
613
614 array_value_type = None
615 for value in lst:
616 dtype = get_dump_type(value)
617 if dtype not in {'b', 'i', 'i64', 'f', 's'}:
618 raise ConfigSerializeError(
619 "Invalid datatype in array (may only contain scalars):"
620 "%r of type %s" % (value, type(value)))
621
622 if array_value_type is None:
623 array_value_type = dtype
624 continue
625
626 if array_value_type == dtype:
627 continue
628
629 if array_value_type == 'i' and dtype == 'i64':
630 array_value_type = 'i64'
631 continue
632
633 if array_value_type == 'i64' and dtype == 'i':
634 continue
635
636 raise ConfigSerializeError(
637 "Mixed types in array (all elements must have same type):"
638 "%r of type %s" % (value, type(value)))
639
640 return array_value_type

643def dump_value(key, value, f, indent=0):
644 '''Save a value of any libconfig type
645
646 This function serializes takes ``key`` and ``value`` and serializes them
647 into ``f``. If ``key`` is ``None``, a list-style output is produced.
648 Otherwise, output has ``key = value`` format.
649 '''
650
651 spaces = ' ' * indent
652
653 if key is None:
654 key_prefix = ''
655 key_prefix_nl = ''
656 else:
657 key_prefix = key + ' = '
658 key_prefix_nl = key + ' =\n' + spaces
659
660 dtype = get_dump_type(value)
661 if dtype == 'd':
662 f.write(u'{}{}{{\n'.format(spaces, key_prefix_nl))
663 dump_dict(value, f, indent + 4)
664 f.write(u'{}}}'.format(spaces))
665 elif dtype == 'l':
666 f.write(u'{}{}(\n'.format(spaces, key_prefix_nl))
667 dump_collection(value, f, indent + 4)
668 f.write(u'\n{})'.format(spaces))
669 elif dtype == 'a':
670 f.write(u'{}{}[\n'.format(spaces, key_prefix_nl))
671 value_dtype = get_array_value_dtype(value)
672
673 # If int array contains one or more Int64, promote all values to i64.
674 if value_dtype == 'i64':
675 value = [LibconfInt64(v) for v in value]
676 dump_collection(value, f, indent + 4)
677 f.write(u'\n{}]'.format(spaces))
678 elif dtype == 's':
679 f.write(u'{}{}{}'.format(spaces, key_prefix, dump_string(value)))
680 elif dtype == 'i' or dtype == 'i64':
681 f.write(u'{}{}{}'.format(spaces, key_prefix, dump_int(value)))
682 elif dtype == 'f' or dtype == 'b':
683 f.write(u'{}{}{}'.format(spaces, key_prefix, value))
684 else:
685 raise ConfigSerializeError("Can not serialize object %r of type %s" %
686 (value, type(value)))
