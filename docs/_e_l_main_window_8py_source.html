106 def connect_to_server(self):
107 """connect to the server with timeout handling"""
108 try:
109 # get connection details
110 self.client.server_ip = self.ip_input.text().strip()
111 self.client.server_port = int(self.port_input.text().strip())
112
113 # establish connection
114 self.client.openConnection()
115
116 # configure socket timeout

121 self.connect_button.setEnabled(False)

123 self.ip_input.setEnabled(False)
124 self.port_input.setEnabled(False)

128 self.save_config()

130 except Exception as e:

146 def send_command(self):
147 """send command with threaded response handling"""
148 if not self.connected:
149 self.output_display.append("not connected to a server.")
150 return
151
152 command = self.command_input.text().strip()
153 if command:
154 try:
155 # send the command
156 self.client.client_socket.sendall(command.encode('utf-8'))
157 self.output_display.append(f"sent: {command}")
158
159 # start response thread
160 response_thread = threading.Thread(target=self.receive_response, args=(command,))
161 response_thread.daemon = True # thread will close with main program
162 response_thread.start()
163
164 self.command_input.clear()
165 except Exception as e:
166 self.output_display.append(f"error sending command: {str(e)}")
167
168 def receive_response(self, command):
169 """handle server response in separate thread with timeout"""
170 try:
171 response = self.client.client_socket.recv(1024).decode()
172 if not response:
173 self.received_signal.emit("server closed connection")
174 self.disconnect_from_server()
175 return
176
177 self.received_signal.emit(response)
178 self.log_command(command, response)
179 self.save_config()
180
181 except socket.timeout:
182 self.received_signal.emit("server response timeout")
183 except ConnectionError:
184 self.received_signal.emit("connection lost")
185 self.disconnect_from_server()
186 except Exception as e:
187 self.received_signal.emit(f"error receiving response: {str(e)}")
188
